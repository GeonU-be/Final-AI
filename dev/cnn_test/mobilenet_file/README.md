이미지 검색 실행 매뉴얼

-----------------------------------------------------------

# 1) 가상환경 활성화 (이미 생성돼 있을 경우)

# .venv 라는 파이썬 가상환경을 사용할 때
source .venv/bin/activate

-----------------------------------------------------------

# 2) Mac 환경(OpenMP 충돌 방지) 설정

# - libomp 중복 로딩으로 인한 OMP Error #15 해결

export KMP_DUPLICATE_LIB_OK=TRUE

-----------------------------------------------------------

# 3) MPS(Fallback) 활성화 (Apple Silicon Mac에서 PyTorch GPU 대체 옵션)
# - M1/M2/M3 칩에서 GPU 대신 Metal(MPS) 사용하도록 설정
# - MPS 사용 불가 시 자동으로 CPU로 폴백(fallback)

export PYTORCH_ENABLE_MPS_FALLBACK=1

-----------------------------------------------------------

# 4) OpenMP 스레드 수 제한 (옵션)
# - 모델이 OpenMP를 많이 사용할 경우 과도한 스레드 점유 방지
# - CPU 발열 및 병렬 처리 문제 완화

export OMP_NUM_THREADS=1

-----------------------------------------------------------

# 5) 모델 실행
# - MobileNetV3 Large 모델 로드 → 사전 데이터 로드 → 검색

python mobilenetv3_search.py

-----------------------------------------------------------
